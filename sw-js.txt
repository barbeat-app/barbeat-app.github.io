const C="barbeat-v197",F=["./","index.html","manifest.webmanifest","icon-180.png","icon-192.png","icon-512.png"];
self.addEventListener("install",e=>{e.waitUntil(caches.open(C).then(c=>c.addAll(F.map(u=>new Request(u,{cache:"reload"})))).then(()=>{if(!self.registration.active)self.skipWaiting()}))});
self.addEventListener("activate",e=>{e.waitUntil(caches.keys().then(k=>Promise.all(k.filter(x=>x!==C).map(x=>caches.delete(x)))).then(()=>self.clients.claim()))});
/* Hors ligne : l'appli s'ouvre depuis le cache (instantané, même sans réseau), puis se met à jour en arrière-plan.
   Seules les ressources de l'appli sont gérées : Supabase, YouTube, Open Food Facts passent directement par le réseau. */
self.addEventListener("fetch",e=>{
 const q=e.request;if(q.method!=="GET")return;let u;try{u=new URL(q.url)}catch(x){return}
 if(u.origin!==self.location.origin||q.headers.has("range"))return;
 const nav=q.mode==="navigate";
 e.respondWith((async()=>{
  const c=await caches.open(C);
  const hit=nav?await c.match("index.html"):await c.match(q,{ignoreSearch:true});
  const net=fetch(q).then(r=>{if(r&&r.ok&&r.type==="basic"){const k=r.clone();if(nav){c.put("index.html",k);c.put(self.registration.scope,r.clone())}else c.put(q,k)}return r}).catch(()=>null);
  if(hit){e.waitUntil(net);return hit}
  const r=await net;return r||(await c.match("index.html"))||Response.error()
 })())});

self.addEventListener("notificationclick",e=>{e.notification.close();e.waitUntil(clients.matchAll({type:"window",includeUncontrolled:true}).then(l=>{const c=l.find(x=>"focus" in x);return c?c.focus():clients.openWindow("./")}))});

self.addEventListener("push",e=>{let d={};try{d=e.data.json()}catch(x){}e.waitUntil(self.registration.showNotification(d.title||"Barbeat",{body:d.body||"Ton point du jour t’attend dans Barbeat.",icon:"icon-192.png",badge:"icon-192.png",tag:d.tag||"barbeat",vibrate:[250,120,250,120,400],silent:false,renotify:true}))});

self.addEventListener("message",e=>{if(e.data==="skip")self.skipWaiting()});
